<?php
/**
 * Ajax API 繧ｨ繝ｳ繝峨・繧､繝ｳ繝・
 * 繝輔Ο繝ｳ繝医お繝ｳ繝峨°繧峨・髱槫酔譛溘Μ繧ｯ繧ｨ繧ｹ繝医ｒ蜃ｦ逅・
 */

require_once 'config.php';

// JSON蠖｢蠑上〒蜃ｺ蜉・
header('Content-Type: application/json; charset=UTF-8');

// CORS蟇ｾ蠢懶ｼ亥ｿ・ｦ√↓蠢懊§縺ｦ・・
header('Access-Control-Allow-Origin: *');
header('Access-Control-Allow-Methods: GET, POST');
header('Access-Control-Allow-Headers: Content-Type');

try {
    // 繝ｪ繧ｯ繧ｨ繧ｹ繝医Γ繧ｽ繝・ラ縺ｮ遒ｺ隱・
    $method = $_SERVER['REQUEST_METHOD'];
    
    if ($method !== 'GET' && $method !== 'POST') {
        throw new Exception('險ｱ蜿ｯ縺輔ｌ縺ｦ縺・↑縺・Μ繧ｯ繧ｨ繧ｹ繝医Γ繧ｽ繝・ラ縺ｧ縺・);
    }
    
    // 繧｢繧ｯ繧ｷ繝ｧ繝ｳ繝代Λ繝｡繝ｼ繧ｿ縺ｮ蜿門ｾ・
    $action = $_GET['action'] ?? $_POST['action'] ?? '';
    
    if (empty($action)) {
        throw new Exception('繧｢繧ｯ繧ｷ繝ｧ繝ｳ縺梧欠螳壹＆繧後※縺・∪縺帙ｓ');
    }
    
    // 繝・・繧ｿ繝吶・繧ｹ謗･邯・
    $database = new Database(DB_HOST, DB_NAME, DB_USER, DB_PASS, DB_CHARSET);
    $deviceManager = new DeviceManager($database);
    
    $response = ['success' => false, 'data' => null, 'message' => ''];
    
    switch ($action) {
        case 'get_services':
            // 繧ｵ繝ｼ繝薙せ蜷堺ｸ隕ｧ繧偵Μ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繝・・繝悶Ν縺九ｉ蜿門ｾ・
            $services = $deviceManager->getServiceNamesFromRelation();
            $response = [
                'success' => true,
                'data' => $services,
                'message' => '繧ｵ繝ｼ繝薙せ蜷堺ｸ隕ｧ繧貞叙蠕励＠縺ｾ縺励◆'
            ];
            break;
            
        case 'get_device_types':
            // 陬・ｽｮ遞ｮ蛻･荳隕ｧ繧偵Μ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繝・・繝悶Ν縺九ｉ蜿門ｾ暦ｼ医し繝ｼ繝薙せ蜷阪〒繝輔ぅ繝ｫ繧ｿ・・
            $serviceName = $_GET['service_name'] ?? $_POST['service_name'] ?? null;
            $deviceTypes = $deviceManager->getDeviceTypesFromRelation($serviceName);
            $response = [
                'success' => true,
                'data' => $deviceTypes,
                'message' => '陬・ｽｮ遞ｮ蛻･荳隕ｧ繧貞叙蠕励＠縺ｾ縺励◆'
            ];
            break;
            
        case 'get_all_relations':
            // 蜈ｨ繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ蜿門ｾ・
            $relations = $deviceManager->getAllRelations();
            $response = [
                'success' => true,
                'data' => $relations,
                'message' => '繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ荳隕ｧ繧貞叙蠕励＠縺ｾ縺励◆'
            ];
            break;
            
        case 'build_relations':
            // 譌｢蟄倥ョ繝ｼ繧ｿ縺九ｉ繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ讒狗ｯ・
            $result = $deviceManager->buildRelationsFromExistingData();
            $response = [
                'success' => $result['success'],
                'data' => null,
                'message' => $result['message']
            ];
            break;
            
        case 'search_devices':
            // 陬・ｽｮ諠・ｱ讀懃ｴ｢
            $serviceName = $_GET['service_name'] ?? $_POST['service_name'] ?? null;
            $deviceType = $_GET['device_type'] ?? $_POST['device_type'] ?? null;
            $deviceName = $_GET['device_name'] ?? $_POST['device_name'] ?? null;
            $page = (int)($_GET['page'] ?? $_POST['page'] ?? 1);
            $limit = 20; // 1繝壹・繧ｸ縺ゅ◆繧翫・莉ｶ謨ｰ
            $offset = ($page - 1) * $limit;
            
            // 遨ｺ譁・ｭ怜・繧地ull縺ｫ螟画鋤
            $serviceName = $serviceName === '' ? null : $serviceName;
            $deviceType = $deviceType === '' ? null : $deviceType;
            $deviceName = $deviceName === '' ? null : $deviceName;
            
            $devices = $deviceManager->searchDevicesAdvanced(
                $serviceName, 
                $deviceType, 
                $deviceName, 
                $limit, 
                $offset
            );
            
            $total = $deviceManager->countDevicesAdvanced(
                $serviceName, 
                $deviceType, 
                $deviceName
            );
            
            $totalPages = ceil($total / $limit);
            
            $response = [
                'success' => true,
                'data' => [
                    'devices' => $devices,
                    'pagination' => [
                        'current_page' => $page,
                        'total_pages' => $totalPages,
                        'total_count' => $total,
                        'per_page' => $limit
                    ]
                ],
                'message' => "讀懃ｴ｢邨先棡: {$total}莉ｶ隕九▽縺九ｊ縺ｾ縺励◆"
            ];
            break;
            
        case 'get_statistics':
            // 陬・ｽｮ邨ｱ險域ュ蝣ｱ繧貞叙蠕・
            $stats = $deviceManager->getDeviceStatistics();
            $response = [
                'success' => true,
                'data' => $stats,
                'message' => '邨ｱ險域ュ蝣ｱ繧貞叙蠕励＠縺ｾ縺励◆'
            ];
            break;
            
        case 'build_relations':
            // 譌｢蟄倥ョ繝ｼ繧ｿ縺九ｉ繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繧定・蜍墓ｧ狗ｯ・
            $buildResult = $deviceManager->buildRelationsFromExistingData();
            $response = [
                'success' => true,
                'data' => $buildResult,
                'message' => "繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ讒狗ｯ牙ｮ御ｺ・ {$buildResult['registered']}莉ｶ逋ｻ骭ｲ"
            ];
            break;
            
        case 'get_all_relations':
            // 蜈ｨ繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ荳隕ｧ繧貞叙蠕暦ｼ育ｮ｡逅・畑・・
            $relations = $deviceManager->getAllRelations();
            $response = [
                'success' => true,
                'data' => $relations,
                'message' => '繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ荳隕ｧ繧貞叙蠕励＠縺ｾ縺励◆'
            ];
            break;
            
        default:
            throw new Exception('荳肴ｭ｣縺ｪ繧｢繧ｯ繧ｷ繝ｧ繝ｳ縺ｧ縺・ ' . $action);
    }
    
} catch (Exception $e) {
    $response = [
        'success' => false,
        'data' => null,
        'message' => $e->getMessage()
    ];
    
    // 繧ｨ繝ｩ繝ｼ繝ｭ繧ｰ縺ｫ險倬鹸・域悽逡ｪ迺ｰ蠅・〒縺ｯ驥崎ｦ・ｼ・
    error_log("Ajax API Error: " . $e->getMessage());
    
    // HTTP繧ｹ繝・・繧ｿ繧ｹ繧ｳ繝ｼ繝峨ｒ險ｭ螳・
    http_response_code(400);
} finally {
    // 繝・・繧ｿ繝吶・繧ｹ謗･邯壹ｒ髢峨§繧・
    if (isset($database)) {
        $database->close();
    }
}

// JSON蠖｢蠑上〒繝ｬ繧ｹ繝昴Φ繧ｹ繧貞・蜉・
echo json_encode($response, JSON_UNESCAPED_UNICODE | JSON_PRETTY_PRINT);

 */
function sanitizeFilename($filename) {
    // 蜊ｱ髯ｺ縺ｪ譁・ｭ励ｒ髯､蜴ｻ
    $filename = preg_replace('/[^a-zA-Z0-9\-_\.]/', '_', $filename);
    // 騾｣邯壹☆繧九ラ繝・ヨ繧・い繝ｳ繝繝ｼ繧ｹ繧ｳ繧｢繧貞腰荳縺ｫ

    return $filename;
}

 * 繝・・繧ｿ繝吶・繧ｹ縺ｮ繝・・繝悶Ν蜷阪→縺励※菴ｿ逕ｨ蜿ｯ閭ｽ縺ｪ譁・ｭ怜・縺ｫ螟画鋤
 * @param string $str
 * @return string
 */
function sanitizeTableName($str) {
    // 譌･譛ｬ隱樊枚蟄励・谿九＠縲∫音谿頑枚蟄励・縺ｿ髯､蜴ｻ
    $str = preg_replace('/[^\p{L}\p{N}_]/u', '_', $str);
    // 騾｣邯壹☆繧九い繝ｳ繝繝ｼ繧ｹ繧ｳ繧｢繧貞腰荳縺ｫ

    // 蜈磯ｭ譛ｫ蟆ｾ縺ｮ繧｢繝ｳ繝繝ｼ繧ｹ繧ｳ繧｢繧帝勁蜴ｻ
    $str = trim($str, '_');
    return $str;
}

/**
 * Docker迺ｰ蠅・・蛛･蜈ｨ諤ｧ繝√ぉ繝・け
 * @return array
 */
function checkDockerEnvironment() {
    $checks = [
        'database_connection' => false,
        'upload_directory' => false,
        'required_extensions' => false
    ];
    
    // 繝・・繧ｿ繝吶・繧ｹ謗･邯壹メ繧ｧ繝・け
    try {
        $database = new Database(DB_HOST, DB_NAME, DB_USER, DB_PASS, DB_CHARSET);
        $database->connect();
        $checks['database_connection'] = true;
    } catch (Exception $e) {
        // 謗･邯壼､ｱ謨・
    }
    
    // 繧｢繝・・繝ｭ繝ｼ繝峨ョ繧｣繝ｬ繧ｯ繝医Μ繝√ぉ繝・け
    if (is_dir(UPLOAD_DIR) && is_writable(UPLOAD_DIR)) {
        $checks['upload_directory'] = true;
    }
    
    // 蠢・ｦ√↑諡｡蠑ｵ讖溯・繝√ぉ繝・け
    if (extension_loaded('pdo') && extension_loaded('pdo_mysql')) {
        $checks['required_extensions'] = true;
    }
    
    return $checks;
}

 * CSV繝倥ャ繝繝ｼ縺九ｉ繧ｫ繝ｩ繝螳夂ｾｩ繧堤函謌・
 */
function generateColumnsFromHeader($header) {
    $columns = [];
    foreach ($header as $columnName) {
        $sanitized = preg_replace('/[^a-zA-Z0-9_]/', '_', $columnName);
        $columns[] = "`{$sanitized}` TEXT";
    }
    return implode(', ', $columns);
}

/**
 * CSV繝輔ぃ繧､繝ｫ蜃ｦ逅・け繝ｩ繧ｹ
 */
class CsvProcessor {
    private $filePath;
    private $headers;
    private $data;
    private $errors;
    
    public function __construct($filePath = null) {
        $this->filePath = $filePath;
        $this->headers = [];
        $this->data = [];
        $this->errors = [];
    }
    
    /**
     * CSV繝輔ぃ繧､繝ｫ繧定ｪｭ縺ｿ霎ｼ繧
     * @param string $filePath
     * @return bool
     */
    public function loadFile($filePath) {
        $this->filePath = $filePath;
        $this->headers = [];
        $this->data = [];
        $this->errors = [];
        
        if (!file_exists($filePath)) {
            $this->errors[] = "繝輔ぃ繧､繝ｫ縺悟ｭ伜惠縺励∪縺帙ｓ: " . $filePath;
            return false;
        }
        
        if (!is_readable($filePath)) {
            $this->errors[] = "繝輔ぃ繧､繝ｫ繧定ｪｭ縺ｿ霎ｼ繧√∪縺帙ｓ: " . $filePath;
            return false;
        }
        
        // 繝輔ぃ繧､繝ｫ縺ｮ繧ｨ繝ｳ繧ｳ繝ｼ繝・ぅ繝ｳ繧ｰ繧呈､懷・繝ｻ螟画鋤
        $content = file_get_contents($filePath);
        $encoding = mb_detect_encoding($content, ['UTF-8', 'SJIS', 'EUC-JP', 'ASCII'], true);
        
        if ($encoding !== 'UTF-8') {
            $content = mb_convert_encoding($content, 'UTF-8', $encoding);
            file_put_contents($filePath, $content);
        }
        
        // CSV繝輔ぃ繧､繝ｫ繧帝幕縺・
        $handle = fopen($filePath, 'r');
        if ($handle === false) {
            $this->errors[] = "CSV繝輔ぃ繧､繝ｫ繧帝幕縺代∪縺帙ｓ";
            return false;
        }
        
        $rowNumber = 0;
        while (($row = fgetcsv($handle, 0, ',')) !== false) {
            $rowNumber++;
            
            if ($rowNumber === 1) {
                // 繝倥ャ繝繝ｼ陦・
                $this->headers = array_map('trim', $row);
                
                // 蠢・医き繝ｩ繝縺ｮ蟄伜惠遒ｺ隱・
                $requiredColumns = ['繧ｵ繝ｼ繝薙せ蜷・, '陬・ｽｮ遞ｮ蛻･', '陬・ｽｮ蜷咲ｧｰ', '繝ｦ繝ｼ繧ｶ繝ｼ蜷・];
                foreach ($requiredColumns as $required) {
                    if (!in_array($required, $this->headers)) {
                        $this->errors[] = "蠢・医き繝ｩ繝縺御ｸ崎ｶｳ縺励※縺・∪縺・ " . $required;
                    }
                }
                
                if (!empty($this->errors)) {
                    fclose($handle);
                    return false;
                }
            } else {
                // 繝・・繧ｿ陦・
                if (count($row) !== count($this->headers)) {
                    $this->errors[] = "陦鶏$rowNumber}: 繧ｫ繝ｩ繝謨ｰ縺御ｸ閾ｴ縺励∪縺帙ｓ・域悄蠕・ " . count($this->headers) . ", 螳滄圀: " . count($row) . "・・;
                    continue;
                }
                
                $rowData = [];
                for ($i = 0; $i < count($this->headers); $i++) {

/**
 * Database謗･邯夂ｮ｡逅・け繝ｩ繧ｹ
 */
class Database {
    private $connection;
    private $host;
    private $dbname;
    private $username;
    private $password;
    private $charset;
    private $inTransaction = false;
    
    public function __construct($host, $dbname, $username, $password, $charset = 'utf8mb4') {
        $this->host = $host;
        $this->dbname = $dbname;
        $this->username = $username;
        $this->password = $password;
        $this->charset = $charset;
    }
    
    /**
     * 繝・・繧ｿ繝吶・繧ｹ縺ｫ謗･邯・
     * @return PDO
     * @throws Exception
     */
    public function connect() {
        if ($this->connection === null) {
            try {
                $dsn = "mysql:host={$this->host};dbname={$this->dbname};charset={$this->charset}";
                $options = [
                    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                    PDO::ATTR_EMULATE_PREPARES => false,
                ];
                
                $this->connection = new PDO($dsn, $this->username, $this->password, $options);
            } catch (PDOException $e) {
                throw new Exception("繝・・繧ｿ繝吶・繧ｹ謗･邯壹お繝ｩ繝ｼ: " . $e->getMessage());
            }
        }
        
        return $this->connection;
    }
    
    /**
     * 謗･邯壹ｒ髢峨§繧・
     */
    public function close() {
        if ($this->connection !== null) {
            // 繧｢繧ｯ繝・ぅ繝悶↑繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ縺後≠繧後・繝ｭ繝ｼ繝ｫ繝舌ャ繧ｯ
            try {
                if ($this->connection->inTransaction()) {
                    error_log("Warning: Active transaction found during connection close, rolling back");
                    $this->connection->rollBack();
                    $this->inTransaction = false;
                }
            } catch (Exception $e) {
                error_log("Error during transaction cleanup: " . $e->getMessage());
            }
            
            $this->connection = null;
        }
    }
    
    /**
     * 繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ髢句ｧ・
     */
    public function beginTransaction() {
        $connection = $this->connect();
        if (!$connection->inTransaction()) {
            $result = $connection->beginTransaction();
            if ($result) {
                $this->inTransaction = true;
            }
            return $result;
        }
        return true; // 譌｢縺ｫ繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ荳ｭ
    }
    
    /**
     * 繧ｳ繝溘ャ繝・
     */
    public function commit() {
        $connection = $this->connect();
        if ($connection->inTransaction()) {
            $result = $connection->commit();
            $this->inTransaction = false;
            return $result;
        }
        return true; // 繧｢繧ｯ繝・ぅ繝悶↑繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ縺後↑縺・
    }
    
    /**
     * 繝ｭ繝ｼ繝ｫ繝舌ャ繧ｯ
     */
    public function rollBack() {
        $connection = $this->connect();
        if ($connection->inTransaction()) {
            $result = $connection->rollBack();
            $this->inTransaction = false;
            return $result;
        }
        return true; // 繧｢繧ｯ繝・ぅ繝悶↑繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ縺後↑縺・
    }
    
    /**
     * 繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ迥ｶ諷九ｒ遒ｺ隱・
     */
    public function inTransaction() {
        return $this->connect()->inTransaction();
    }
    
    /**
     * 繝励Μ繝壹い繝峨せ繝・・繝医Γ繝ｳ繝医・螳溯｡・
     * @param string $query
     * @param array $params
     * @return PDOStatement
     */
    public function execute($query, $params = []) {
        try {
            $stmt = $this->connect()->prepare($query);
            $stmt->execute($params);
            return $stmt;
        } catch (PDOException $e) {
            throw new Exception("繧ｯ繧ｨ繝ｪ螳溯｡後お繝ｩ繝ｼ: " . $e->getMessage() . " SQL: " . $query);
        }
    }
    
    /**
     * 繝・・繝悶Ν縺ｮ蟄伜惠遒ｺ隱・
     * @param string $tableName
     * @return bool
     */
    public function tableExists($tableName) {
        try {
            $stmt = $this->execute(
                "SELECT COUNT(*) FROM information_schema.tables WHERE table_schema = ? AND table_name = ?",
                [$this->dbname, $tableName]
            );
            return $stmt->fetchColumn() > 0;
        } catch (Exception $e) {
            throw new Exception("繝・・繝悶Ν蟄伜惠遒ｺ隱阪お繝ｩ繝ｼ: " . $e->getMessage());
        }
    }
    
    /**
     * 繝・・繝悶Ν縺ｮ繧ｫ繝ｩ繝諠・ｱ繧貞叙蠕・
     * @param string $tableName
     * @return array
     */
    public function getTableColumns($tableName) {
        try {
            $stmt = $this->execute(
                "SELECT COLUMN_NAME, DATA_TYPE, IS_NULLABLE, COLUMN_DEFAULT 
                 FROM information_schema.columns 
                 WHERE table_schema = ? AND table_name = ? 
                 ORDER BY ORDINAL_POSITION",
                [$this->dbname, $tableName]
            );
            return $stmt->fetchAll();
        } catch (Exception $e) {
            throw new Exception("繧ｫ繝ｩ繝諠・ｱ蜿門ｾ励お繝ｩ繝ｼ: " . $e->getMessage());
        }
    }
}

     * @throws Exception
     */
    public function insertOrUpdateDynamicData($tableName, $data) {
        $tableName = sanitizeTableName($tableName);
        
        if (empty($data)) {
            return true;
        }
        
        // 繧ｫ繝ｩ繝蜷阪→繝励Ξ繝ｼ繧ｹ繝帙Ν繝繝ｼ繧呈ｺ門ｙ
        $columns = [];
        $placeholders = [];
        $updateClauses = [];
        $params = [];
        $placeholderIndex = 0;
        
        foreach ($data as $key => $value) {
            // 繧ｫ繝ｩ繝蜷阪・譌･譛ｬ隱槭・縺ｾ縺ｾ菴ｿ逕ｨ・医ヰ繝・け繧ｯ繧ｩ繝ｼ繝医〒繧ｨ繧ｹ繧ｱ繝ｼ繝暦ｼ・
            $columns[] = "`{$key}`";
            
            // 繝励Ξ繝ｼ繧ｹ繝帙Ν繝繝ｼ蜷阪・闍ｱ謨ｰ蟄励・縺ｿ・・aram0, param1, ...・・
            $placeholder = "param" . $placeholderIndex;
            $placeholders[] = ":{$placeholder}";
            $params[$placeholder] = $value;
            
            // 荳ｻ繧ｭ繝ｼ莉･螟悶・譖ｴ譁ｰ蜿･繧剃ｽ懈・
            $primaryKeyColumns = array_keys($data);
            if ($key !== $primaryKeyColumns[0]) {
                $updateClauses[] = "`{$key}` = VALUES(`{$key}`)";
            }
            
            $placeholderIndex++;
        }
        
        $sql = "
            INSERT INTO `{$tableName}` 
            (" . implode(", ", $columns) . ")
            VALUES (" . implode(", ", $placeholders) . ")
        ";
        
        if (!empty($updateClauses)) {
            $sql .= " ON DUPLICATE KEY UPDATE " . implode(", ", $updateClauses) . ", updated_at = CURRENT_TIMESTAMP";
        }
        
        try {
            error_log("Dynamic insert SQL: " . $sql);
            error_log("Dynamic insert params: " . json_encode($params, JSON_UNESCAPED_UNICODE));
            
            $this->database->execute($sql, $params);
            return true;
        } catch (Exception $e) {
            error_log("Dynamic insert error: " . $e->getMessage());
            error_log("SQL: " . $sql);
            error_log("Params: " . json_encode($params, JSON_UNESCAPED_UNICODE));
            throw new Exception("蜍慕噪繝・・繝悶Ν '{$tableName}' 縺ｸ縺ｮ繝・・繧ｿ謖ｿ蜈･縺ｫ螟ｱ謨励＠縺ｾ縺励◆: " . $e->getMessage());
        }
    }
    
    /**
     * CSV繝・・繧ｿ繧剃ｸ諡ｬ蜃ｦ逅・
     * @param CsvProcessor $csvProcessor
     * @return array 蜃ｦ逅・ｵ先棡
     * @throws Exception
     */
    public function processCsvData(CsvProcessor $csvProcessor) {
        $results = [
            'success' => false,
            'device_info_count' => 0,
            'dynamic_tables_created' => [],
            'dynamic_data_count' => 0,
            'errors' => []
        ];
        
        try {
            // 繝医Λ繝ｳ繧ｶ繧ｯ繧ｷ繝ｧ繝ｳ髢句ｧ・
            $this->database->beginTransaction();
            
            // 陬・ｽｮ諠・ｱ繝・・繝悶Ν縺悟ｭ伜惠縺励↑縺・ｴ蜷医・菴懈・
            if (!$this->deviceInfoTableExists()) {
                $this->createDeviceInfoTable();
            }
            
            $data = $csvProcessor->getData();
            $extendedColumns = $csvProcessor->getExtendedColumns();
            
            // 菴懈・縺悟ｿ・ｦ√↑蜍慕噪繝・・繝悶Ν繧堤音螳・
            $dynamicTables = [];
            foreach ($data as $row) {
                $tableName = $csvProcessor->generateTableName($row);
                if (!isset($dynamicTables[$tableName])) {
                    $dynamicTables[$tableName] = true;
                }
            }
            
            // 蜍慕噪繝・・繝悶Ν繧剃ｽ懈・
            foreach (array_keys($dynamicTables) as $tableName) {
                if (!$this->dynamicTableExists($tableName)) {
                    // 荳ｻ繧ｭ繝ｼ繧ｫ繝ｩ繝蜷阪ｒ豁｣縺励￥逕滓・
                    $primaryKeyColumn = $csvProcessor->generatePrimaryKeyColumnName($data[0]);
                    $this->createDynamicTable($tableName, $primaryKeyColumn, $extendedColumns);
                    $results['dynamic_tables_created'][] = $tableName;
                }
            }
            
            // 繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繝・・繝悶Ν縺悟ｭ伜惠縺励↑縺・ｴ蜷医・菴懈・
            if (!$this->relationTableExists()) {
                $this->createRelationTable();
            }
            
            // 繝・・繧ｿ繧貞・逅・
            foreach ($data as $row) {
                // 陬・ｽｮ諠・ｱ繝・・繝悶Ν縺ｫ謖ｿ蜈･
                $deviceInfo = $csvProcessor->convertToDeviceInfo($row);
                $this->insertOrUpdateDeviceInfo($deviceInfo);
                $results['device_info_count']++;
                
                // 繧ｵ繝ｼ繝薙せ蜷阪→陬・ｽｮ遞ｮ蛻･縺ｮ繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繧堤匳骭ｲ
                try {
                    $this->registerServiceDeviceTypeRelation(
                        $row['繧ｵ繝ｼ繝薙せ蜷・],
                        $row['陬・ｽｮ遞ｮ蛻･'],
                        'CSV閾ｪ蜍慕匳骭ｲ'
                    );
                } catch (Exception $e) {
                    // 繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ逋ｻ骭ｲ繧ｨ繝ｩ繝ｼ縺ｯ繝ｭ繧ｰ縺ｫ險倬鹸縺吶ｋ縺悟・逅・・邯咏ｶ・
                    error_log("繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ逋ｻ骭ｲ繧ｨ繝ｩ繝ｼ: " . $e->getMessage());
                }
                
                // 蜍慕噪繝・・繝悶Ν縺ｫ謖ｿ蜈･・域僑蠑ｵ繧ｫ繝ｩ繝縺悟ｭ伜惠縺吶ｋ蝣ｴ蜷医・縺ｿ・・
                if (!empty($extendedColumns)) {
                    $tableName = $csvProcessor->generateTableName($row);
                    $extendedData = $csvProcessor->convertToExtendedData($row);
                    $this->insertOrUpdateDynamicData($tableName, $extendedData);
                    $results['dynamic_data_count']++;
                }
            }
            
            // 繧ｳ繝溘ャ繝・
            $this->database->commit();
            $results['success'] = true;
            
        } catch (Exception $e) {
            // 繝ｭ繝ｼ繝ｫ繝舌ャ繧ｯ
            $this->database->rollBack();
            $results['errors'][] = $e->getMessage();
            throw $e;
        }
        
        return $results;
    }

                is_active = 1,

            'service_name' => $serviceName,
            'device_type' => $deviceType,

        $sql = "

     * @param string $serviceName
     * @param string $deviceType
     * @return bool
     */
    public function checkRelationExists($serviceName, $deviceType) {
        // 繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繝・・繝悶Ν縺悟ｭ伜惠縺励↑縺・ｴ蜷医・蟶ｸ縺ｫtrue繧定ｿ斐☆
        if (!$this->relationTableExists()) {
            return true;
        }
        
        $sql = "
            SELECT COUNT(*) 
            FROM service_device_type_relations
            WHERE service_name = :service_name 
            AND device_type = :device_type 
            AND is_active = 1
        ";
        
        $params = [
            'service_name' => $serviceName,
            'device_type' => $deviceType
        ];
        
        try {
            $stmt = $this->database->execute($sql, $params);
            return $stmt->fetchColumn() > 0;
        } catch (Exception $e) {
            throw new Exception("繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ蟄伜惠遒ｺ隱阪↓螟ｱ謨励＠縺ｾ縺励◆: " . $e->getMessage());
        }
    }
    
    /**
     * 譌｢蟄倥・device_info縺九ｉ繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繧定・蜍墓ｧ狗ｯ・
     * @return array 讒狗ｯ臥ｵ先棡
     * @throws Exception
     */
    public function buildRelationsFromExistingData() {
        // 繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ繝・・繝悶Ν縺ｮ菴懈・
        if (!$this->relationTableExists()) {
            $this->createRelationTable();
        }
        
        // 譌｢蟄倥・繧ｵ繝ｼ繝薙せ蜷阪→陬・ｽｮ遞ｮ蛻･縺ｮ邨・∩蜷医ｏ縺帙ｒ蜿門ｾ・
        $sql = "
            SELECT DISTINCT 
                service_name, 
                device_type,
                COUNT(*) as device_count
            FROM device_info 
            GROUP BY service_name, device_type
            ORDER BY service_name, device_type
        ";
        
        try {
            $stmt = $this->database->execute($sql);
            $combinations = $stmt->fetchAll();
            
            $registered = 0;
            $errors = [];
            
            foreach ($combinations as $combination) {
                try {
                    $description = "陬・ｽｮ謨ｰ: {$combination['device_count']}蜿ｰ (閾ｪ蜍墓ｧ狗ｯ・";
                    $this->registerServiceDeviceTypeRelation(
                        $combination['service_name'],
                        $combination['device_type'],
                        $description
                    );
                    $registered++;
                } catch (Exception $e) {
                    $errors[] = "{$combination['service_name']} - {$combination['device_type']}: " . $e->getMessage();
                }
            }
            
            return [
                'success' => true,
                'registered' => $registered,
                'total_combinations' => count($combinations),
                'errors' => $errors
            ];
            
        } catch (Exception $e) {
            throw new Exception("繝ｪ繝ｬ繝ｼ繧ｷ繝ｧ繝ｳ閾ｪ蜍墓ｧ狗ｯ峨↓螟ｱ謨励＠縺ｾ縺励◆: " . $e->getMessage());
        }
    }
